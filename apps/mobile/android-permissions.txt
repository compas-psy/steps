# compas-psy/zapiski, apps/mobile/android-permissions.txt; разбор —
# .ultraplan/research/04-android-release.md).
#
# Формат: имя разрешения, при необходимости `maxSdkVersion=N`, затем `#` и
# объяснение — зачем оно и что сломается без него. Строка без объяснения не
# проходит проверку (`test/android-permissions.test.ts`): разрешение без
# причины — это разрешение, которое никто не пересмотрит.
#
android.permission.INTERNET
  # Пересмотрено (было «сознательно ОТСУТСТВУЕТ» до этой строки — история в
  # git). Причина пересмотра — не продуктовое решение, а факт рантайма:
  # `INTERNET` объявляет не ШАГИ и не какая-то зависимость, а сам сгенерированный
  # базовый манифест Tauri (`apps/mobile/src-tauri/gen/android/app/src/main/
  # AndroidManifest.xml:3:5-67`, до любого патча этого workflow) — подтверждено
  # `manifest-merger-<variant>-report.txt`: `ADDED from .../AndroidManifest.xml`,
  # не от зависимости и не от нашего кода. WebView-мост Tauri требует его
  # безусловно (загрузка `asset://`/`tauri://` через встроенный сервер), APK
  # без него не соберётся штатным `tauri android build`. Реальная приватность
  # локального режима (SPEC/00 §9) держится не на отсутствии разрешения в
  # манифесте (оно физически не даёт приложению отправить данные без кода,
  # который сеть использует), а на architecture-boundary в коде: до аккаунта/
  # синка R1b НИ ОДИН путь в приложении не вызывает сеть — `usesCleartextTraffic
  # ="false"` там же в манифесте дополнительно ограничивает даже теоретическую
  # попытку до HTTPS. Раньше документированное «физически не может выйти в
  # сеть» было неверным утверждением о рантайме — это тоже причина пересмотра:
  # проверяемое обещание должно быть верным, а не удобным.

android.permission.POST_NOTIFICATIONS
  # Android 13+ требует его для ЛЮБОГО показа уведомления, включая локальные
  # напоминания задач (SPEC/00 §0 — «local notifications» в целях R1). Без
  # строки уведомление молча не показывается на 13+, а разрешение не
  # появляется в системном диалоге вовсе.

android.permission.SCHEDULE_EXACT_ALARM
  # Точное время напоминания, которое человек указал явно (SPEC/00 §11.1 —
  # «Android: explicit timed reminders use native scheduling»). С Android 12
  # система различает точные и неточные будильники; без этого разрешения
  # `AlarmManager.setExactAndAllowWhileIdle` бросает исключение, а не
  # тихо деградирует до неточного — деградацию обязан явно показать UI
  # (SPEC §11.1: «never silently present an inexact alarm as exact»), а не
  # получить её как побочный эффект нехватки разрешения. НЕ вместе с
  # `USE_EXACT_ALARM` (Android документирует их как взаимоисключающий выбор
  # для одного приложения, не пару «оба сразу»): `USE_EXACT_ALARM` рассчитан
  # на приложения, для которых точный будильник — основное назначение
  # (будильник/таймер/календарь), выдаётся автоматически, но жёстче
  # ограничен политикой стора; ШАГИ — планировщик задач с напоминанием как
  # одной из функций, не будильник, поэтому `SCHEDULE_EXACT_ALARM` — та же
  # модель, что уже выбрана: человек при необходимости даёт особый доступ
  # вручную, а приложение проверяет `canScheduleExactAlarms()` и явно
  # показывает деградацию, если доступа нет (см. абзац выше).

android.permission.RECEIVE_BOOT_COMPLETED
  # Пересоздание расписания уведомлений после перезагрузки устройства
  # (`02_DATA_MODEL_SYNC.md` §14 — «Every startup/background wake compares
  # desired schedule fingerprints with OS scheduled notification IDs»):
  # `AlarmManager` не переживает reboot, все точные будильники слетают
  # молча. Без этого разрешения boot receiver не запускается вовсе, и
  # напоминания перестают срабатывать после каждой перезагрузки без единого
  # сообщения об ошибке.

android.permission.VIBRATE
  # Хэптика (`src/platform.ts` → `createHaptics`, Web Vibration API в
  # системном WebView). Без строки `navigator.vibrate()` тихо возвращает
  # `false` и ничего не происходит — SecurityException не бросается, но и
  # обратной связи у человека тоже нет.

android.permission.WAKE_LOCK
  # Обнаружено CI-шлюзом («Разрешения собранного APK против списка»), не
  # решением продукта — как и `INTERNET` выше. Объявлено безусловно самим
  # `tauri-plugin-notification` 2.4.0 (ADR-0008) в его собственном
  # `android/src/main/AndroidManifest.xml` (`<uses-permission
  # android:name="android.permission.WAKE_LOCK" />`) — подтверждено прямым
  # чтением исходника пакета, не догадкой. В Kotlin-коде самого плагина нет
  # ни одного вызова `PowerManager`/`WakeLock` API (проверено — grep по
  # всему `android/` пакета пуст): это защитное объявление библиотеки для
  # класса задач «получатель, разбуженный AlarmManager, должен успеть
  # показать уведомление раньше, чем устройство снова уснёт», а не решение,
  # которое принял или мог отключить наш код. Без строки штатная сборка
  # `tauri android build` не даст собрать APK этим CI-шлюзом вовсе — тот же
  # случай, что и с `INTERNET`.

ru.cmpas.shagi.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION
  # НЕ пользовательское разрешение и не сторонняя зависимость — саморазрешение
  # с `protectionLevel="signature"`, которое AndroidX Core объявляет и сам же
  # выдаёт приложению для безопасной реализации `Context.registerReceiver`
  # с `RECEIVER_NOT_EXPORTED` на API, где системный флаг ещё недоступен.
  # `signature` означает «выдаётся только пакету, подписанному тем же
  # ключом» — постороннее приложение получить его не может, в системном
  # диалоге установки не показывается вовсе. Имя вычисляется из
  # `applicationId` (`ru.cmpas.shagi`, зафиксирован в `tauri.conf.json`) —
  # штатная инфраструктура AndroidX, применяется у любого современного
  # Android-приложения с динамически регистрируемым receiver где-либо в
  # дереве зависимостей, не специфична для ШАГОВ.
